cmake_minimum_required(VERSION 3.18)
project(M3TControlledAblation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ENGINE_ROOT "${COMPARISON_KIT}/sources/current_m3t")
set(USE_ONNXRUNTIME ON)
set(USE_AZURE_KINECT OFF)
set(USE_REALSENSE OFF)
set(USE_GTEST OFF)
set(OpenGL_GL_PREFERENCE LEGACY)
find_library(ONNXRUNTIME_LIBRARY NAMES onnxruntime PATHS "${ONNXRUNTIME_DIR}/lib" NO_DEFAULT_PATH REQUIRED)
find_package(Eigen3 3.3.2 REQUIRED)
find_package(OpenGL REQUIRED)
find_package(GLEW REQUIRED)
find_package(glfw3 REQUIRED)
find_package(OpenCV 4.3 REQUIRED COMPONENTS core imgproc highgui imgcodecs calib3d features2d video xfeatures2d)
find_package(OpenMP REQUIRED)
include_directories("${CMAKE_CURRENT_SOURCE_DIR}/build_inputs" "${ENGINE_ROOT}/include" "${ENGINE_ROOT}/third_party" ${OpenCV_INCLUDE_DIRS})
add_subdirectory("${ENGINE_ROOT}/src" "engine")
# Override only this translation unit inside this independent build. The
# original frozen library, T-LESS and all prior executables remain untouched.
get_target_property(M3T_SOURCES m3t SOURCES)
set(M3T_SOURCES_BEFORE "${M3T_SOURCES}")
list(FILTER M3T_SOURCES EXCLUDE REGEX "(^|/)texture_modality\\.cpp$")
if("${M3T_SOURCES}" STREQUAL "${M3T_SOURCES_BEFORE}")
  message(FATAL_ERROR "Cannot find the frozen texture_modality.cpp source")
endif()
set_property(TARGET m3t PROPERTY SOURCES "${M3T_SOURCES}")
target_sources(m3t PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/build_inputs/texture_modality.cpp")
add_executable(evaluate_ablation "build_inputs/ablation.cpp" "build_inputs/rbot_evaluator.cpp")
target_link_libraries(m3t PUBLIC OpenMP::OpenMP_CXX)
target_link_libraries(evaluate_ablation PRIVATE m3t OpenMP::OpenMP_CXX)
target_compile_options(m3t PRIVATE $<$<CONFIG:Release>:-O3>)
target_compile_options(evaluate_ablation PRIVATE $<$<CONFIG:Release>:-O3>)
