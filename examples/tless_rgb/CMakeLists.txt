cmake_minimum_required(VERSION 3.18)
project(TLESSRGBRobustness LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ENGINE_ROOT "${COMPARISON_KIT}/sources/current_m3t")
set(USE_ONNXRUNTIME ON)
set(USE_AZURE_KINECT OFF)
set(USE_REALSENSE OFF)
set(USE_GTEST OFF)
set(OpenGL_GL_PREFERENCE LEGACY)
find_library(ONNXRUNTIME_LIBRARY NAMES onnxruntime PATHS "${ONNXRUNTIME_DIR}/lib" NO_DEFAULT_PATH REQUIRED)
find_package(Eigen3 3.3.2 REQUIRED)
find_package(OpenGL REQUIRED)
find_package(GLEW REQUIRED)
find_package(glfw3 REQUIRED)
find_package(OpenCV 4.3 REQUIRED COMPONENTS core imgproc highgui imgcodecs calib3d features2d video xfeatures2d)
find_package(OpenMP REQUIRED)
include_directories(BEFORE "${CMAKE_CURRENT_SOURCE_DIR}/build_inputs/override" "${CMAKE_CURRENT_SOURCE_DIR}")
include_directories("${ENGINE_ROOT}/include" "${ENGINE_ROOT}/third_party" ${OpenCV_INCLUDE_DIRS})
add_subdirectory("${ENGINE_ROOT}/src" "engine")
add_executable(evaluate_tless_rgb "build_inputs/tless.cpp")
target_compile_definitions(evaluate_tless_rgb PRIVATE BENCH_OURS=1)
target_link_libraries(m3t PUBLIC OpenMP::OpenMP_CXX)
target_link_libraries(evaluate_tless_rgb PRIVATE m3t OpenMP::OpenMP_CXX)
target_compile_options(m3t PRIVATE $<$<CONFIG:Release>:-O3>)
target_compile_options(evaluate_tless_rgb PRIVATE $<$<CONFIG:Release>:-O3>)
add_executable(test_rgb_geometry "rgb_geometry_test.cpp")
target_link_libraries(test_rgb_geometry PRIVATE ${OpenCV_LIBRARIES})
