cmake_minimum_required(VERSION 3.18)
project(IndustrialTrackerComparison LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BENCH_METHOD "M3T" CACHE STRING "RBGT, SRT3D, ICG or M3T")
set(BASELINE_ROOT "" CACHE PATH "Unmodified 3DObjectTracking source snapshot")
if(NOT BENCH_METHOD MATCHES "^(RBGT|SRT3D|ICG|M3T|OURS)$")
  message(FATAL_ERROR "Unknown BENCH_METHOD")
endif()
set(ENGINE_ROOT "${BASELINE_ROOT}/${BENCH_METHOD}")
if(BENCH_METHOD STREQUAL "OURS")
  set(ENGINE_ROOT "${CURRENT_M3T_ROOT}")
  set(USE_ONNXRUNTIME ON)
  find_library(ONNXRUNTIME_LIBRARY NAMES onnxruntime PATHS "${ONNXRUNTIME_DIR}/lib" NO_DEFAULT_PATH REQUIRED)
endif()
if(NOT EXISTS "${ENGINE_ROOT}/src/CMakeLists.txt")
  message(FATAL_ERROR "Missing baseline source: ${ENGINE_ROOT}")
endif()
set(USE_AZURE_KINECT OFF)
set(USE_REALSENSE OFF)
set(USE_GTEST OFF)
set(OpenGL_GL_PREFERENCE LEGACY)
find_package(Eigen3 3.3.2 REQUIRED)
find_package(OpenGL REQUIRED)
find_package(GLEW REQUIRED)
find_package(glfw3 REQUIRED)
find_package(OpenCV 4.3 REQUIRED COMPONENTS core imgproc highgui imgcodecs calib3d features2d video xfeatures2d)
find_package(OpenMP REQUIRED)
include_directories("${ENGINE_ROOT}/include" "${ENGINE_ROOT}/third_party" ${OpenCV_INCLUDE_DIRS})
# Compile the supplied algorithms' own source lists, without their demo/device
# applications. The algorithm source files themselves remain unmodified.
add_subdirectory("${ENGINE_ROOT}/src" "engine")
string(TOLOWER "${BENCH_METHOD}" ENGINE_TARGET)
if(BENCH_METHOD STREQUAL "OURS")
  set(ENGINE_TARGET m3t)
endif()
target_link_libraries(${ENGINE_TARGET} PUBLIC OpenMP::OpenMP_CXX)
if(BENCH_METHOD STREQUAL "OURS")
  add_executable(evaluate_industrial_baseline "${ENGINE_ROOT}/examples/evaluate_industrial_dataset.cpp" "${ENGINE_ROOT}/examples/rbot_evaluator.cpp")
else()
  add_executable(evaluate_industrial_baseline baseline.cpp)
endif()
target_compile_definitions(evaluate_industrial_baseline PRIVATE "BENCH_${BENCH_METHOD}=1")
target_link_libraries(evaluate_industrial_baseline PRIVATE ${ENGINE_TARGET} OpenMP::OpenMP_CXX)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()
target_compile_options(${ENGINE_TARGET} PRIVATE $<$<CONFIG:Release>:-O3>)
target_compile_options(evaluate_industrial_baseline PRIVATE $<$<CONFIG:Release>:-O3>)
