cmake_minimum_required(VERSION 3.18)
project(RBOTFiveMethods LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT BENCH_METHOD MATCHES "^(RBGT|SRT3D|ICG|M3T|OURS)$")
  message(FATAL_ERROR "Invalid BENCH_METHOD")
endif()
set(ENGINE_ROOT "${COMPARISON_KIT}/sources/upstream/${BENCH_METHOD}")
if(BENCH_METHOD STREQUAL "OURS")
  set(ENGINE_ROOT "${COMPARISON_KIT}/sources/current_m3t")
  set(USE_ONNXRUNTIME ON)
  find_library(ONNXRUNTIME_LIBRARY NAMES onnxruntime PATHS "${ONNXRUNTIME_DIR}/lib" NO_DEFAULT_PATH REQUIRED)
endif()
set(USE_AZURE_KINECT OFF)
set(USE_REALSENSE OFF)
set(USE_GTEST OFF)
set(OpenGL_GL_PREFERENCE LEGACY)
find_package(Eigen3 3.3.2 REQUIRED)
find_package(OpenGL REQUIRED)
find_package(GLEW REQUIRED)
find_package(glfw3 REQUIRED)
find_package(OpenCV 4.3 REQUIRED COMPONENTS core imgproc highgui imgcodecs calib3d features2d video xfeatures2d)
find_package(OpenMP REQUIRED)
include_directories("${ENGINE_ROOT}/include" "${ENGINE_ROOT}/third_party" "${ENGINE_ROOT}/examples" ${OpenCV_INCLUDE_DIRS})
add_subdirectory("${ENGINE_ROOT}/src" "engine")
string(TOLOWER "${BENCH_METHOD}" ENGINE_TARGET)
if(BENCH_METHOD STREQUAL "OURS")
  set(ENGINE_TARGET m3t)
  add_executable(evaluate_rbot_comparison "${CMAKE_CURRENT_SOURCE_DIR}/build_inputs/rbot_ours.cpp" "${ENGINE_ROOT}/examples/rbot_evaluator.cpp")
else()
  add_executable(evaluate_rbot_comparison "${CMAKE_CURRENT_SOURCE_DIR}/build_inputs/rbot_baseline.cpp")
endif()
target_compile_definitions(evaluate_rbot_comparison PRIVATE "BENCH_${BENCH_METHOD}=1")
target_link_libraries(${ENGINE_TARGET} PUBLIC OpenMP::OpenMP_CXX)
target_link_libraries(evaluate_rbot_comparison PRIVATE ${ENGINE_TARGET} OpenMP::OpenMP_CXX)
target_compile_options(${ENGINE_TARGET} PRIVATE $<$<CONFIG:Release>:-O3>)
target_compile_options(evaluate_rbot_comparison PRIVATE $<$<CONFIG:Release>:-O3>)
